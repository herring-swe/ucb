280static inline size_t ucb_buffer_grow_double(ucb_buffer* buf, size_t size_needed)
281{
282 size_t cap = buf->capacity * 2;
283 while (size_needed > cap - buf->used)
284 cap *= 2;
285 return cap;
286}
