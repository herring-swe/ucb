164/*
165 * Allocation with explicit type, using pointer type.
166 * Must be used AFTER ptr have been declared.
167 *
168 * Note: Requires C23 or later or Visual Studio 19.39 or later.
169 * Due to this, it's not used internally.
170 *
171 * Usage example ():
172 * struct MyType *my_type;
173 * my_type = ucb_calloc_ptr(1, my_type);
174 */
175#if defined(__STDC_VERSION__) && __STDC_VERSION__ >= 202311L
176#define ucb_malloc_ptr(ptr, num) ((typeof(ptr))ucb_malloc((num) * sizeof(*(ptr))))
177#define ucb_calloc_ptr(ptr, num) ((typeof(ptr))ucb_calloc((num), sizeof(*(ptr))))
178#define ucb_realloc_ptr(ptr, num) ((typeof(ptr))ucb_realloc((ptr), (num) * sizeof(*(ptr))))
179#elif defined(__gcc__) || defined(__clang__) || (defined(_MSC_VER) && _MSC_VER >= 1939)
180#define ucb_malloc_ptr(ptr, num) ((__typeof__(ptr))ucb_malloc((num) * sizeof(*(ptr))))
